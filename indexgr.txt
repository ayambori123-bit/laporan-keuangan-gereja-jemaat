        // Add Pemasukan with Automatic Calculation Rules
        function handleTambahPemasukan(e) {
            e.preventDefault();
            const date = document.getElementById('inDate').value;
            const kategori = document.getElementById('inKategori').value;
            const uraian = document.getElementById('inUraian').value;
            const nominal = parseFloat(document.getElementById('inNominal').value) || 0;

            // ATURAN OTOMATIS:
            // 1. Perpuluhan = 100% Sinode
            // 2. Diakonia, Pembangunan = 100% Jemaat
            // 3. Lainnya / Umum (Pundi, Syukur, PAR, Tunas, PAM, PW, Rayon 1 & 2, PKB, Perjamuan Kudus): Klasis 20%, Sinode 40%, Jemaat 40%
            let klasis = 0;
            let sinode = 0;
            let jemaat = 0;

            if(kategori === 'Perpuluhan') {
                sinode = nominal;
            } else if(kategori === 'Diakonia' || kategori === 'Pembangunan') {
                jemaat = nominal;
            } else {
                klasis = nominal * 0.20;
                sinode = nominal * 0.40;
                jemaat = nominal * 0.40;
            }

            const item = {
                id: Date.now(),
                date,
                kategori,
                uraian,
                nominal,
                klasis,
                sinode,
                jemaat
            };

            dataPemasukan.push(item);
            localStorage.setItem('gki_pemasukan', JSON.stringify(dataPemasukan));
            
            document.getElementById('pemasukanForm').reset();
            document.getElementById('inDate').value = new Date().toISOString().split('T')[0];
            
            renderAll();
            showToast('Pemasukan berhasil disimpan dan dihitung otomatis!', 'success');
        }

                if(item.kategori === 'Perpuluhan') totalPerpuluhan += item.nominal;
                totalKlasis += item.klasis;
                totalSinodeAll += item.sinode;
                totalJemaatAll += item.jemaat;

                tbodyIn.innerHTML += `
                    <tr class="hover:bg-slate-50 transition border-b border-slate-100">
                        <td class="p-4 text-slate-600 font-medium">${item.date}</td>
                        <td class="p-4"><span class="px-2.5 py-1 ${item.kategori === 'Perpuluhan' ? 'bg-amber-100 text-amber-900 border border-amber-300' : 'bg-gki-50 text-gki-700'} rounded-lg text-xs font-bold">${item.kategori}</span></td>
                        <td class="p-4 text-slate-800">${item.uraian}</td>
                        <td class="p-4 text-right font-bold text-slate-900">Rp ${item.nominal.toLocaleString('id-ID')}</td>
                        <td class="p-4 text-center text-xs text-indigo-600 font-medium">Rp ${item.klasis.toLocaleString('id-ID')}</td>
                        <td class="p-4 text-center text-xs text-blue-600 font-medium">Rp ${item.sinode.toLocaleString('id-ID')}</td>
                        <td class="p-4 text-center text-xs text-emerald-600 font-medium">Rp ${item.jemaat.toLocaleString('id-ID')}</td>
                        <td class="p-4 text-center">
                            <button onclick="hapusPemasukan(${item.id})" class="p-2 bg-slate-100 hover:bg-rose-50 text-slate-400 hover:text-rose-600 rounded-lg transition cursor-pointer" title="Hapus">
                                <i class="fas fa-trash-alt"></i>
                            </button>
                        </td>
                    </tr>
                `;
            });

            if(dataPemasukan.length === 0) {
                tbodyIn.innerHTML = `<tr><td colspan="8" class="p-8 text-center text-slate-400">Belum ada data pemasukan tercatat.</td></tr>`;
            }

            // Render Pengeluaran Table
            const tbodyOut = document.getElementById('listPengeluaranBody');
            tbodyOut.innerHTML = '';
            let totalPengeluaran = 0;

            dataPengeluaran.forEach(item => {
                totalPengeluaran += item.nominal;
                tbodyOut.innerHTML += `
                    <tr class="hover:bg-slate-50 transition border-b border-slate-100">
                        <td class="p-4 text-slate-600 font-medium">${item.date}</td>
                        <td class="p-4"><span class="px-2.5 py-1 bg-rose-50 text-rose-700 rounded-lg text-xs font-semibold">${item.kategori}</span></td>
                        <td class="p-4 text-slate-800">${item.uraian}</td>
                        <td class="p-4 text-right font-bold text-rose-600">Rp ${item.nominal.toLocaleString('id-ID')}</td>
                        <td class="p-4 text-center">
                            ${item.buktiName !== 'Tidak ada bukti' ? `<a href="${item.buktiUrl}" target="_blank" class="text-xs text-gki-700 font-semibold underline hover:text-gki-800"><i class="fas fa-file-alt mr-1"></i>${item.buktiName}</a>` : `<span class="text-xs text-slate-400">Tanpa Bukti</span>`}
                        </td>
                        <td class="p-4 text-center">
                            <button onclick="hapusPengeluaran(${item.id})" class="p-2 bg-slate-100 hover:bg-rose-50 text-slate-400 hover:text-rose-600 rounded-lg transition cursor-pointer" title="Hapus">
                                <i class="fas fa-trash-alt"></i>
                            </button>
                        </td>
                    </tr>
                `;
            });

            if(dataPengeluaran.length === 0) {
                tbodyOut.innerHTML = `<tr><td colspan="6" class="p-8 text-center text-slate-400">Belum ada data pengeluaran tercatat.</td></tr>`;
            }

            // Update Counts in tabs
            document.getElementById('countIn').innerText = dataPemasukan.length;
            document.getElementById('countOut').innerText = dataPengeluaran.length;

            // Hitung Kas Jemaat Aktif (Total Alokasi Jemaat - Total Pengeluaran)
            const kasJemaatAktif = totalJemaatAll - totalPengeluaran;

            // Update Cards Dashboard
            document.getElementById('cardTotalPemasukan').innerText = `Rp ${totalPemasukan.toLocaleString('id-ID')}`;
            document.getElementById('cardPerpuluhan').innerText = `Rp ${totalPerpuluhan.toLocaleString('id-ID')}`;
            document.getElementById('cardKasJemaat').innerText = `Rp ${kasJemaatAktif.toLocaleString('id-ID')}`;
            document.getElementById('cardTotalPengeluaran').innerText = `Rp ${totalPengeluaran.toLocaleString('id-ID')}`;
            document.getElementById('cardKlasis').innerText = `Rp ${totalKlasis.toLocaleString('id-ID')}`;
            document.getElementById('cardSinode').innerText = `Rp ${totalSinodeAll.toLocaleString('id-ID')}`;
            document.getElementById('cardJemaatAlokasi').innerText = `Rp ${totalJemaatAll.toLocaleString('id-ID')}`;

            // Update Breakdown Sinode text
            const sinodeUmum = totalSinodeAll - totalPerpuluhan;
            document.getElementById('sinodeBreakdown').innerText = `Rp ${sinodeUmum.toLocaleString('id-ID')} (40%) + Rp ${totalPerpuluhan.toLocaleString('id-ID')} (Perpuluhan)`;
        }

        // Copy to clipboard helper
        function copyToClipboard(text, label) {
            const dummy = document.createElement("textarea");
            document.body.appendChild(dummy);
            dummy.value = text;
            dummy.select();
            document.execCommand("copy");
            document.body.removeChild(dummy);
            showToast(`${label} berhasil disalin!`, 'success');
        }

        // Modal Print Toggle & Render Report Content
        function bukaModalCetak() {
            let totalIn = dataPemasukan.reduce((acc, curr) => acc + curr.nominal, 0);
            let totalOut = dataPengeluaran.reduce((acc, curr) => acc + curr.nominal, 0);
            let totalPerp = dataPemasukan.filter(i => i.kategori === 'Perpuluhan').reduce((acc, curr) => acc + curr.nominal, 0);
            let totalJemaat = dataPemasukan.reduce((acc, curr) => acc + curr.jemaat, 0);
            let kasAktif = totalJemaat - totalOut;

            document.getElementById('repTotalIn').innerText = `Rp ${totalIn.toLocaleString('id-ID')}`;
            document.getElementById('repTotalOut').innerText = `Rp ${totalOut.toLocaleString('id-ID')}`;
            document.getElementById('repPerpuluhan').innerText = `Rp ${totalPerp.toLocaleString('id-ID')}`;
            document.getElementById('repKas').innerText = `Rp ${kasAktif.toLocaleString('id-ID')}`;

            const options = { year: 'numeric', month: 'long', day: 'numeric' };
            document.getElementById('printCurrentDate').innerText = new Date().toLocaleDateString('id-ID', options);

            const printInBody = document.getElementById('printTableInBody');
            printInBody.innerHTML = '';
            dataPemasukan.forEach(item => {
                printInBody.innerHTML += `
                    <tr class="border-b border-slate-200">
                        <td class="p-1.5 border-r border-slate-200">${item.date}</td>
                        <td class="p-1.5 border-r border-slate-200">${item.kategori}</td>
                        <td class="p-1.5 border-r border-slate-200">${item.uraian}</td>
                        <td class="p-1.5 text-right font-bold">Rp ${item.nominal.toLocaleString('id-ID')}</td>
                    </tr>
                `;
            });
            if(dataPemasukan.length === 0) printInBody.innerHTML = `<tr><td colspan="4" class="p-2 text-center text-slate-400">Tidak ada data</td></tr>`;

            const printOutBody = document.getElementById('printTableOutBody');
            printOutBody.innerHTML = '';
            dataPengeluaran.forEach(item => {
                printOutBody.innerHTML += `
                    <tr class="border-b border-slate-200">
                        <td class="p-1.5 border-r border-slate-200">${item.date}</td>
                        <td class="p-1.5 border-r border-slate-200">${item.kategori}</td>
                        <td class="p-1.5 border-r border-slate-200">${item.uraian}</td>
                        <td class="p-1.5 text-right font-bold text-rose-600">Rp ${item.nominal.toLocaleString('id-ID')}</td>

            document.getElementById('printModal').classList.remove('hidden');
        }
